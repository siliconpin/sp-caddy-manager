2. Repeated Database Transaction Pattern
Every handler repeats this pattern:

go
tx, err := a.DB.Begin()
committed := false
defer func() {
    if !committed {
        tx.Rollback()
    }
}()
// ... operations ...

    // cleanup on failure
    return
}
committed = true
